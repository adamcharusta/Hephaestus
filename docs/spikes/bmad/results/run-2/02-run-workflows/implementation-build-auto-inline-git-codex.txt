## shelf/inventory.py (50 lines)
| """In-memory inventory of a small shop."""
| 
| 
| class Inventory:
|     """Quantities of items, by name."""
| 
|     def __init__(self):
|         self._items = {}
| 
|     def add(self, name, quantity):
|         """Add `quantity` of the item `name`."""
|         _check_quantity(quantity)
|         self._items[name] = self._items.get(name, 0) + quantity
| 
|     def remove(self, name, quantity):
|         """Remove `quantity` of the item `name`.
| 
|         An item whose quantity reaches zero stays in the inventory.
|         """
|         _check_quantity(quantity)
|         if quantity > self._items.get(name, 0):
|             raise ValueError(f"not enough of {name!r} in stock")
|         self._items[name] -= quantity
| 
|     def quantity(self, name):
|         """The quantity of the item `name`; zero for an unknown item."""
|         return self._items.get(name, 0)
| 
|     def names(self):
|         """The names of all items, sorted."""
|         return sorted(self._items)
| 
|     def low_stock(self, threshold):
|         """Return the names of items whose quantity is at or below `threshold`.
| 
|         The result is sorted alphabetically. A negative `threshold` raises
|         `ValueError`. The method treats an item with quantity `0` as present
|         in the inventory and includes it when appropriate.
|         """
|         # Validate threshold: must be an int and not negative. Reject booleans.
|         if isinstance(threshold, bool) or not isinstance(threshold, int) or threshold < 0:
|             raise ValueError("threshold must be a non-negative integer")
| 
|         names = [name for name, qty in self._items.items() if qty <= threshold]
|         return sorted(names)
| 
| 
| def _check_quantity(quantity):
|     if isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0:
|         raise ValueError("quantity must be a positive integer")
